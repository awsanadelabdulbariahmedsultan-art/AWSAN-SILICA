def calculate_awsan_silica_roi(annual_tonnage=500000):
    # Dynamic distribution of product value-addition chains
    raw_sand_revenue = (annual_tonnage * 0.40) * 20
    refined_powder_revenue = (annual_tonnage * 0.35) * 250

    # Financial constants

    total_opex = 45000000    # Annualized Operational Costs
    
    net_profit = total_revenue - total_opex
    roi_percentage = (net_profit / total_capex) * 100
    payback_period = total_capex / net_profit if net_profit > 0 else float('inf')
    
    print("====================================================================")
    print(f"💵 AWSAN SILICA FINANCIAL PROJECTIONS FOR {annual_tonnage:,} TONS/YEAR")
    print("====================================================================")
    print(f"• Total Projected Revenue: \${total_revenue:,}")
    print(f"• Total Capital Investment (CAPEX): \${total_capex:,}")
    print(f"• Total Annual Operations (OPEX): \${total_opex:,}")
    print(f"• Net Annual Profit for Awsan Dew: \${net_profit:,}")
    print(f"• Return on Investment (ROI): {roi_percentage:.2f}%")
    print(f"• Project Capital Payback Period: {payback_period:.2f} Years")
    print("====================================================================")

# Execute core simulation engine
